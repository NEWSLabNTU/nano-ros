# integrations/s32ds/CMakeLists.txt
#
# RFC-0062 — NXP S32 Design Studio integration shell.
#
# S32DS is an ecosystem that owns its own build: the CDT project already has
# the kernel, RTD drivers, startup, vector table, linker scripts and pin/clock
# config. So nano-ros integrates the way it does with ESP-IDF — it does NOT
# supply a board. It builds its own objects with the project's exact ABI and
# hands them back as static libraries for the CDT link.
#
# Division of labour:
#
#   S32DS builds ......... FreeRTOS kernel, RTD drivers, startup, vector table,
#                          clock/pin config, and links the final .elf
#   this shell builds .... nros-platform-freertos shim, the RMW, the generated
#                          node/message code, the app entry
#   YOU supply ........... a TCP/IP stack (see the LWIP_DIR block below)
#
# Usage:
#   cmake -S <nano-ros>/integrations/s32ds -B build-s32ds \
#         -DNXP_S32DS_PROJECT=~/MR_CANHUBK3_IEEE1722 \
#         -DNROS_WORKSPACE_DIR=~/my_ws -DNROS_BRINGUP=safety_island_bringup \
#         -DLWIP_DIR=<lwip source>
#   cmake --build build-s32ds
#   cp build-s32ds/nros-libs.mk  ~/MR_CANHUBK3_IEEE1722/
#   cp <nano-ros>/integrations/s32ds/makefile.defs ~/MR_CANHUBK3_IEEE1722/
#   make -C ~/MR_CANHUBK3_IEEE1722/Debug_FLASH all
#
# STATUS: skeleton. The probe is validated against the MR-CANHUBK3 reference
# project; the full link has NOT been exercised on hardware. See RFC-0062.

cmake_minimum_required(VERSION 3.22)

# ---------------------------------------------------------------------------
# Probe the S32DS project BEFORE project() — it is pure file I/O and needs no
# compiler, and its output is what the toolchain file is made of.
# ---------------------------------------------------------------------------
include("${CMAKE_CURRENT_LIST_DIR}/cmake/S32dsProject.cmake")

if(NOT NXP_S32DS_PROJECT)
    set(NXP_S32DS_PROJECT "$ENV{NXP_S32DS_PROJECT}")
endif()
if(NOT NXP_S32DS_PROJECT)
    message(FATAL_ERROR
        "nros-s32ds: set -DNXP_S32DS_PROJECT=<S32DS project root> (or the "
        "NXP_S32DS_PROJECT env var) — the directory holding .cproject.")
endif()

nros_s32ds_probe(PROJECT "${NXP_S32DS_PROJECT}" CONFIG "${NXP_S32DS_CONFIG}")

# ---------------------------------------------------------------------------
# Generate the toolchain file from what the project actually uses.
#
# This is deliberately NOT a checked-in `cmake/toolchain/arm-freertos-armcm7.cmake`.
# A static toolchain file would restate the project's ABI by hand and could
# drift from it silently — the exact failure RFC-0062 flags in the mps2 CM3
# toolchain file, which fuses arch with `-fno-exceptions -fno-rtti -std=c++14`.
# Derived-from-the-project cannot drift.
# ---------------------------------------------------------------------------
string(JOIN " " _abi_str ${NROS_S32DS_ABI_FLAGS} ${NROS_S32DS_SPECS})
set(_tc "${CMAKE_BINARY_DIR}/s32ds-toolchain.cmake")
file(WRITE "${_tc}" "\
# GENERATED by integrations/s32ds — derived from ${NXP_S32DS_PROJECT}
# config ${NROS_S32DS_CONFIG}. Do not edit; re-run cmake instead.
set(CMAKE_SYSTEM_NAME      Generic)
set(CMAKE_SYSTEM_PROCESSOR arm)
set(CMAKE_C_COMPILER   arm-none-eabi-gcc)
set(CMAKE_CXX_COMPILER arm-none-eabi-g++)
set(CMAKE_ASM_COMPILER arm-none-eabi-gcc)
set(CMAKE_C_FLAGS_INIT   \"${_abi_str} -ffunction-sections -fdata-sections\")
set(CMAKE_CXX_FLAGS_INIT \"${_abi_str} -ffunction-sections -fdata-sections\")
set(CMAKE_ASM_FLAGS_INIT \"${_abi_str}\")
set(Rust_CARGO_TARGET \"thumbv7em-none-eabihf\" CACHE STRING \"\" FORCE)
set(CMAKE_FIND_ROOT_PATH_MODE_PROGRAM NEVER)
set(CMAKE_FIND_ROOT_PATH_MODE_LIBRARY ONLY)
set(CMAKE_FIND_ROOT_PATH_MODE_INCLUDE ONLY)
# Bare-metal idiom (issue 0366): ABI detection must not try to LINK an
# executable, or newlib's unresolved syscall stubs abort configure.
set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY)
")
if(NOT CMAKE_TOOLCHAIN_FILE)
    set(CMAKE_TOOLCHAIN_FILE "${_tc}")
endif()

project(nros_s32ds LANGUAGES C CXX)

get_filename_component(_nros_root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)

# ---------------------------------------------------------------------------
# freertos_kernel — HEADERS ONLY, deliberately.
#
# S32DS already compiles every kernel source into the image (its makefile
# includes FreeRTOS/Source/subdir.mk and the ARM_CM7 port). Declaring the
# sources here too would put a second copy of `xTaskCreate` & friends in the
# link. The shim only ever needs the headers + FreeRTOSConfig.h.
# ---------------------------------------------------------------------------
add_library(freertos_kernel INTERFACE)
target_include_directories(freertos_kernel INTERFACE ${NROS_S32DS_INCLUDE_DIRS})
if(NROS_S32DS_FREERTOS_DIR)
    target_include_directories(freertos_kernel INTERFACE
        "${NROS_S32DS_FREERTOS_DIR}/include"
        "${NROS_S32DS_FREERTOS_DIR}/portable/${NROS_S32DS_FREERTOS_PORT}")
endif()
foreach(_extra IN LISTS NROS_S32DS_EXTRA_INCLUDES)
    target_include_directories(freertos_kernel INTERFACE "${_extra}")
endforeach()

# Hand the shim's own CMake the same facts.
set(FREERTOS_DIR    "${NROS_S32DS_FREERTOS_DIR}"  CACHE PATH   "" FORCE)
set(FREERTOS_PORT   "${NROS_S32DS_FREERTOS_PORT}" CACHE STRING "" FORCE)

# ---------------------------------------------------------------------------
# lwip — YOUR responsibility, and nano-ros will not pretend otherwise.
#
# The MR-CANHUBK3 FreeRTOS deliverable ships NO TCP/IP stack: `src/enet.c`
# drives the GMAC directly for raw IEEE-1722/AVTP frames. nano-ros's zenoh and
# Cyclone backends both want sockets, so something has to provide them.
# ---------------------------------------------------------------------------
if(NOT LWIP_DIR)
    message(FATAL_ERROR
        "nros-s32ds: -DLWIP_DIR=<lwip source> is required.\n"
        "The S32DS FreeRTOS deliverable has no TCP/IP stack (src/enet.c drives\n"
        "the GMAC raw for AVTP), so you must choose one of:\n"
        "  1. lwIP + a netif driver over the vendor MAC — `nros setup --source\n"
        "     lwip`, then write <chip>_lwip.c (see packages/drivers/net/\n"
        "     lan9118-lwip/src/lan9118_lwip.c, ~507 LOC, as the reference).\n"
        "  2. Implement the socket subset of the nros_platform_* ABI directly\n"
        "     over the vendor MAC (packages/platform/nros-platform-api).\n"
        "  3. Declare LinkFeatures{custom} and supply a transport vtable —\n"
        "     smallest surface, but no TCP/UDP locators.\n"
        "Set -DLWIP_DIR to your stack's source root for (1).")
endif()

# lwIP is not usable from its source tree alone: `lwip/opt.h` does
# `#include "lwipopts.h"` and the FreeRTOS port needs `arch/cc.h`. Both are
# per-deployment files (buffer counts, which protocols, MEM_SIZE …), so they
# are yours, not nano-ros's. Reference pair:
#   packages/boards/nros-board-freertos/config/lwipopts.h
#   packages/boards/nros-board-freertos/config/arch/cc.h
if(NOT LWIP_CONFIG_DIR)
    message(FATAL_ERROR
        "nros-s32ds: -DLWIP_CONFIG_DIR=<dir> is required — a directory "
        "containing `lwipopts.h` (and `arch/cc.h` for the FreeRTOS port).\n"
        "lwIP cannot compile without them and their values are a property of "
        "YOUR deployment, not of nano-ros. Start from\n"
        "  packages/boards/nros-board-freertos/config/ (phase-337 W5.a moved the\n"
        "  shared defaults there; the mps2 board keeps only its own numbers)\n"
        "and size MEM_SIZE / PBUF / TCP buffers for the S32K344's 320 KB SRAM "
        "— the mps2 values assume a QEMU target with far more RAM.")
endif()

add_library(lwip INTERFACE)
target_include_directories(lwip INTERFACE
    "${LWIP_CONFIG_DIR}"
    "${LWIP_DIR}/src/include"
    "${LWIP_DIR}/contrib/ports/freertos/include")

# ---------------------------------------------------------------------------
# Compose `freertos_platform` BEFORE add_subdirectory(<nano-ros>).
#
# This is what puts `cmake/platform/nano-ros-freertos.cmake` into board-less
# mode (RFC-0062 revision 1): it sees the umbrella already exists and skips the
# per-board overlay entirely, so this board contributes ZERO files to the
# nano-ros tree.
# ---------------------------------------------------------------------------
add_library(freertos_platform INTERFACE)
target_link_libraries(freertos_platform INTERFACE freertos_kernel lwip)

set(NANO_ROS_PLATFORM "freertos" CACHE STRING "" FORCE)
if(NOT NANO_ROS_RMW)
    set(NANO_ROS_RMW "zenoh" CACHE STRING "" FORCE)
endif()

# `packages/api/nros-{c,cpp}/CMakeLists.txt` do
# `include("${NANO_ROS_ROOT}/cmake/NanoRosFeatureSet.cmake")`, and the root
# CMakeLists sets `NANO_ROS_ROOT_DIR` — a DIFFERENT variable. Setting
# NANO_ROS_ROOT is the consumer's job by convention: `nano_ros_workspace()`,
# `NanoRosBootstrap.cmake`, `zephyr/CMakeLists.txt` and
# `integrations/px4/NanoRosPx4Module.cmake` each do it. Without this the
# configure fails with `include could not find requested file:
# /cmake/NanoRosFeatureSet.cmake` (empty prefix).
set(NANO_ROS_ROOT "${_nros_root}" CACHE PATH "nano-ros repo root" FORCE)

add_subdirectory("${_nros_root}" nano_ros_root)

# ---------------------------------------------------------------------------
# The image's ending (issue 0719). This shell builds the nros-c/nros-cpp
# staticlibs without a `nano_ros_entry()`, so it applies the policy itself —
# without it the `no_std` staticlib has no `#[panic_handler]` and does not
# compile (issue 1664 measured that; the shell had never been built through).
# `platform` routes a panic to `nros_platform_panic`, the FreeRTOS shim's.
# ---------------------------------------------------------------------------
nros_apply_panic_policy(platform "integrations/s32ds")

# ---------------------------------------------------------------------------
# Which ONE runtime archive the image links (issue 1664, the NuttX fix of
# issue 1636).
#
# `libnros_cpp.a` bundles nros-c (nros-cpp's `rmw-<x>-cffi` features forward
# `nros-c/rmw-<x>`), so it defines every `nros_*` C-ABI symbol `libnros_c.a`
# does. Linking BOTH puts two different nros-c builds in one image — every
# `nros_*`, the QoS constants and `REGISTRY`, the backend registry, twice, each
# copy with its own statics. NuttX measured 392 such duplicates. So the image
# takes exactly one: `libnros_cpp.a` for a C++-API image, `libnros_c.a`
# otherwise. Never both, and never a glob that could catch both.
# ---------------------------------------------------------------------------
set(NROS_S32DS_API "C" CACHE STRING
    "nano-ros API the S32DS image calls: C (libnros_c.a) or CPP (libnros_cpp.a)")
set_property(CACHE NROS_S32DS_API PROPERTY STRINGS C CPP)
string(TOUPPER "${NROS_S32DS_API}" _nros_api)
if(_nros_api STREQUAL "CPP")
    set(_nros_runtime_target nros_cpp-static)
elseif(_nros_api STREQUAL "C")
    set(_nros_runtime_target nros_c-static)
else()
    message(FATAL_ERROR
        "nros-s32ds: NROS_S32DS_API='${NROS_S32DS_API}' — expected C or CPP.")
endif()
if(NOT TARGET ${_nros_runtime_target})
    message(FATAL_ERROR
        "nros-s32ds: NROS_S32DS_API=${_nros_api} needs the Rust target "
        "${_nros_runtime_target}, which this configure did not import.")
endif()

# ---------------------------------------------------------------------------
# Emit `nros-libs.mk` — the CDT link fragment.
#
# Mirrors integrations/nuttx's `Make.defs`. `makefile.defs` (dropped into the
# S32DS project root) includes this and appends to USER_OBJS, which the CDT
# link rule already passes to gcc verbatim.
# ---------------------------------------------------------------------------
set(_libs_mk "${CMAKE_BINARY_DIR}/nros-libs.mk")
file(GENERATE OUTPUT "${_libs_mk}" CONTENT
"# GENERATED by integrations/s32ds — do not edit.
# Source project: ${NXP_S32DS_PROJECT} (${NROS_S32DS_CONFIG})
NROS_BUILD_DIR := ${CMAKE_BINARY_DIR}

# The C platform shim.
USER_OBJS += $<TARGET_FILE:nros_platform_freertos>

# The ONE Rust runtime archive (NROS_S32DS_API=${_nros_api}). See the comment
# above NROS_S32DS_API in integrations/s32ds/CMakeLists.txt: nros-cpp bundles
# nros-c, so the two are alternatives, never a pair (issue 1664).
USER_OBJS += $<TARGET_FILE:${_nros_runtime_target}>

# User hook for further archives. A Rust staticlib added here (e.g. a
# per-package message FFI archive) carries a second copy of the Rust runtime
# and is a duplicate-symbol link error — the build has no
# duplicate-symbol relaxation to hide it (issues 1645, 1664).
-include \$(NROS_BUILD_DIR)/nros-extra-libs.mk
")

message(STATUS "nros-s32ds: will emit ${_libs_mk} (runtime: ${_nros_runtime_target})")
